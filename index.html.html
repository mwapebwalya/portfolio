# ETL + scheduling module
import pandas as pd
from apscheduler import BlockingScheduler

def transform(raw):
    df = pd.read_csv(raw)
    df = df.dropna().groupby('category').sum()
    return df

sched = BlockingScheduler()
sched.add_job(transform, 'cron', hour=2)
sched.start()
